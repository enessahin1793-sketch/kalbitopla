<script src="https://cdn.jsdelivr.net/npm/three@0.152.2/build/three.min.js"></script>

<script>
/* =======================
   SAHNE
======================= */
const scene = new THREE.Scene();
scene.background = new THREE.Color(0xffc0cb);

const camera = new THREE.PerspectiveCamera(
  75,
  window.innerWidth / window.innerHeight,
  0.1,
  1000
);
camera.position.set(0, 5, 10);
camera.lookAt(0, 1, 0);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(window.innerWidth, window.innerHeight);
document.body.appendChild(renderer.domElement);

/* =======================
   IŞIK
======================= */
const light = new THREE.DirectionalLight(0xffffff, 1);
light.position.set(5, 10, 5);
scene.add(light);
scene.add(new THREE.AmbientLight(0xffffff, 0.6));

/* =======================
   ZEMİN
======================= */
const ground = new THREE.Mesh(
  new THREE.PlaneGeometry(50, 1000),
  new THREE.MeshStandardMaterial({ color: 0x90ee90 })
);
ground.rotation.x = -Math.PI / 2;
ground.position.z = -400;
scene.add(ground);

/* =======================
   KIZ KARAKTER (BASİT MODEL)
======================= */
const girl = new THREE.Group();

// Gövde
const body = new THREE.Mesh(
  new THREE.CylinderGeometry(0.5, 0.6, 2),
  new THREE.MeshStandardMaterial({ color: 0xff69b4 })
);
body.position.y = 1;
girl.add(body);

// Kafa
const head = new THREE.Mesh(
  new THREE.SphereGeometry(0.5),
  new THREE.MeshStandardMaterial({ color: 0xffe0bd })
);
head.position.y = 2.3;
girl.add(head);

// Saç
const hair = new THREE.Mesh(
  new THREE.SphereGeometry(0.55),
  new THREE.MeshStandardMaterial({ color: 0x000000 })
);
hair.position.y = 2.45;
hair.scale.y = 0.8;
girl.add(hair);

girl.position.z = 5;
scene.add(girl);

/* =======================
   KALPLER
======================= */
const hearts = [];
const heartGeo = new THREE.SphereGeometry(0.4, 16, 16);
const heartMat = new THREE.MeshStandardMaterial({ color: 0xff0000 });

function spawnHeart() {
  const heart = new THREE.Mesh(heartGeo, heartMat);
  heart.position.set(
    (Math.random() - 0.5) * 6,
    1,
    -Math.random() * 100 - 20
  );
  scene.add(heart);
  hearts.push(heart);
}

for (let i = 0; i < 15; i++) spawnHeart();

/* =======================
   KONTROLLER (MOBİL + PC)
======================= */
let targetX = 0;

function movePlayer(x) {
  targetX = Math.max(-5, Math.min(5, x));
}

window.addEventListener("touchmove", e => {
  const x = (e.touches[0].clientX / window.innerWidth - 0.5) * 12;
  movePlayer(x);
});

window.addEventListener("mousemove", e => {
  if (e.buttons === 1) {
    const x = (e.clientX / window.innerWidth - 0.5) * 12;
    movePlayer(x);
  }
});

/* =======================
   OYUN DÖNGÜSÜ
======================= */
function animate() {
  requestAnimationFrame(animate);

  // Karakter hareket
  girl.position.x += (targetX - girl.position.x) * 0.1;

  // Kalpler
  hearts.forEach(h => {
    h.position.z += 0.4;

    // Toplama
    if (h.position.distanceTo(girl.position) < 1) {
      h.position.z = -Math.random() * 100 - 50;
      h.position.x = (Math.random() - 0.5) * 6;
    }

    // Arkadan gelirse
    if (h.position.z > 10) {
      h.position.z = -Math.random() * 100 - 50;
    }
  });

  renderer.render(scene, camera);
}

animate();

/* =======================
   RESPONSIVE
======================= */
window.addEventListener("resize", () => {
  camera.aspect = window.innerWidth / window.innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(window.innerWidth, window.innerHeight);
});
</script>
